CREATE DATABASE WebApartment;
USE WebApartment;

-- ================= ROLE =================
CREATE TABLE VaiTro (
    maVaiTro VARCHAR(100) PRIMARY KEY,
    tenVaiTro VARCHAR(50)
);

-- ================= USER =================
CREATE TABLE NguoiDung (
    maNguoiDung VARCHAR(100) PRIMARY KEY,
    maVaiTro VARCHAR(100),
    hoVaTen VARCHAR(255),
    email VARCHAR(255),
    diaChi VARCHAR(255),
    soDienThoai VARCHAR(30),
    trangThai BOOLEAN,
    matKhau VARCHAR(255),
    facebookAccount VARCHAR(255),
    googleAccount VARCHAR(255),
    anhDaiDien VARCHAR(255),

    FOREIGN KEY (maVaiTro) REFERENCES VaiTro(maVaiTro)
);

-- ================= DANH MUC =================
CREATE TABLE DanhMuc (
    maDanhMuc VARCHAR(100) PRIMARY KEY,
    tenDanhMuc VARCHAR(255)
);

-- ================= BAI DANG =================
CREATE TABLE BaiDang (
    maBaiDang VARCHAR(100) PRIMARY KEY,
    maNguoiDung VARCHAR(100),
    maDanhMuc VARCHAR(100),
    tieuDe VARCHAR(255),
    noiDung TEXT,
    ngayDang DATE,
    trangThai VARCHAR(255),
    lienHe VARCHAR(50),
    phuongThucThanhToan VARCHAR(255),

    FOREIGN KEY (maNguoiDung) REFERENCES NguoiDung(maNguoiDung),
    FOREIGN KEY (maDanhMuc) REFERENCES DanhMuc(maDanhMuc)
);

-- ================= CHI TIET CAN HO =================
CREATE TABLE ChiTietCanHo (
    maChiTietCanHo VARCHAR(100) PRIMARY KEY,
    maBaiDang VARCHAR(100),
    gia DOUBLE,
    dienTich FLOAT,
    phongNgu INT,
    diaChiCuThe VARCHAR(255),
    huongCanHo VARCHAR(255),
    phuong VARCHAR(255),
    lat DOUBLE,
    lng DOUBLE,
    ngayTao DATETIME,

    FOREIGN KEY (maBaiDang) REFERENCES BaiDang(maBaiDang)
);

-- ================= TIEN ICH =================
CREATE TABLE TienIch (
    maTienIch VARCHAR(100) PRIMARY KEY,
    tenTienIch VARCHAR(255)
);

-- ================= BAI DANG - TIEN ICH =================
CREATE TABLE BaiDang_TienIch (
    maBaiDang VARCHAR(100),
    maTienIch VARCHAR(100),
    PRIMARY KEY (maBaiDang, maTienIch),

    FOREIGN KEY (maBaiDang) REFERENCES BaiDang(maBaiDang),
    FOREIGN KEY (maTienIch) REFERENCES TienIch(maTienIch)
);

-- ================= HINH ANH =================
CREATE TABLE HinhAnhBaiDang (
    maHinhAnhBaiDang VARCHAR(100) PRIMARY KEY,
    maBaiDang VARCHAR(100),
    loai VARCHAR(255),
    duongDan VARCHAR(255),
    thumbnailUrl VARCHAR(255),
    thuTu INT,

    FOREIGN KEY (maBaiDang) REFERENCES BaiDang(maBaiDang)
);

-- ================= YEU THICH =================
CREATE TABLE BaiDangYeuThich (
    maNguoiDung VARCHAR(100),
    maBaiDang VARCHAR(100),
    ngayTao DATETIME,
    PRIMARY KEY (maNguoiDung, maBaiDang),

    FOREIGN KEY (maNguoiDung) REFERENCES NguoiDung(maNguoiDung),
    FOREIGN KEY (maBaiDang) REFERENCES BaiDang(maBaiDang)
);

-- ================= NHU CAU USER (AI) =================
CREATE TABLE NhuCauNguoiDung (
    maNhuCauNguoiDung VARCHAR(100) PRIMARY KEY,
    maNguoiDung VARCHAR(100),
    minPrice DOUBLE,
    maxPrice DOUBLE,
    phuong VARCHAR(50),
    loaiCanHo VARCHAR(50),
    coBanCong BOOLEAN,
    dayDuNoiThat BOOLEAN,
    ganTrungTam BOOLEAN,
    ganBien BOOLEAN,
    ngayTao DATETIME,

    FOREIGN KEY (maNguoiDung) REFERENCES NguoiDung(maNguoiDung)
);

-- ================= LICH SU TIM KIEM =================
CREATE TABLE LichSuTimKiem (
    maLichSuTimKiem VARCHAR(100) PRIMARY KEY,
    maNguoiDung VARCHAR(100),
    tuKhoa VARCHAR(255),
    minPrice DOUBLE,
    maxPrice DOUBLE,
    phuong VARCHAR(255),
    thoiGian DATETIME,

    FOREIGN KEY (maNguoiDung) REFERENCES NguoiDung(maNguoiDung)
);

-- ================= TUONG TAC USER =================
CREATE TABLE TuongTacNguoiDung (
    maTuongTacNguoiDung VARCHAR(100) PRIMARY KEY,
    maNguoiDung VARCHAR(100),
    maBaiDang VARCHAR(100),
    loaiHanhDong VARCHAR(255),
    ngayTao DATETIME,

    FOREIGN KEY (maNguoiDung) REFERENCES NguoiDung(maNguoiDung),
    FOREIGN KEY (maBaiDang) REFERENCES BaiDang(maBaiDang)
);

-- ================= PHUONG THUC THANH TOAN =================
CREATE TABLE PhuongThucThanhToan (
    maPhuongThucThanhToan VARCHAR(100) PRIMARY KEY,
    tenPhuongThucThanhToan VARCHAR(255),
    moTa VARCHAR(255)
);

-- ================= MUC DICH THANH TOAN =================
CREATE TABLE MucDichThanhToan (
    maMucDichThanhToan VARCHAR(100) PRIMARY KEY,
    tenMucDich VARCHAR(255)
);

-- ================= GIAO DICH =================
CREATE TABLE GiaoDich (
    maGiaoDich VARCHAR(100) PRIMARY KEY,
    maNguoiDung VARCHAR(100),
    maThanhToan VARCHAR(100),
    maMucDichThanhToan VARCHAR(100),
    soTien DOUBLE,
    trangThai VARCHAR(255),
    ngayTao DATETIME,
    noiDung VARCHAR(255),

    FOREIGN KEY (maNguoiDung) REFERENCES NguoiDung(maNguoiDung),
    FOREIGN KEY (maThanhToan) REFERENCES PhuongThucThanhToan(maPhuongThucThanhToan),
    FOREIGN KEY (maMucDichThanhToan) REFERENCES MucDichThanhToan(maMucDichThanhToan)
);

-- ================= HOA DON =================
CREATE TABLE HoaDonThueCanHo (
    maHoaDon VARCHAR(100) PRIMARY KEY,
    maNguoiDung VARCHAR(100),
    maMucDichThanhToan VARCHAR(100),
    ngayLapHoaDon DATETIME,
    tongTien DOUBLE,
    trangThai VARCHAR(50),
    ghiChu VARCHAR(255),

    FOREIGN KEY (maNguoiDung) REFERENCES NguoiDung(maNguoiDung),
    FOREIGN KEY (maMucDichThanhToan) REFERENCES MucDichThanhToan(maMucDichThanhToan)
);

-- ================= CHI TIET HOA DON =================
CREATE TABLE ChiTietHoaDon (
    maChiTietHoaDon VARCHAR(100) PRIMARY KEY,
    maHoaDon VARCHAR(100),
    maBaiDang VARCHAR(100),
    noiDung VARCHAR(255),
    soTien DOUBLE,
    ngayTao DATETIME,

    FOREIGN KEY (maHoaDon) REFERENCES HoaDonThueCanHo(maHoaDon),
    FOREIGN KEY (maBaiDang) REFERENCES BaiDang(maBaiDang)
);

-- ================= CHI TIET THANH TOAN =================
CREATE TABLE ChiTietThanhToan (
    maChiTietThanhToan VARCHAR(100) PRIMARY KEY,
    maNguoiDung VARCHAR(100),
    maThanhToan VARCHAR(100),
    ngayBatDau DATE,
    ngayKetThuc DATE,
    soTien DOUBLE,
    trangThai VARCHAR(255),
    ngayTao DATETIME,

    FOREIGN KEY (maNguoiDung) REFERENCES NguoiDung(maNguoiDung),
    FOREIGN KEY (maThanhToan) REFERENCES PhuongThucThanhToan(maPhuongThucThanhToan)
);
