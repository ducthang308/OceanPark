CREATE TABLE VaiTro (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tenVaiTro VARCHAR(50)
) ENGINE=InnoDB;

CREATE TABLE NguoiDung (
    id INT AUTO_INCREMENT PRIMARY KEY,
    idVaiTro INT,
    hoVaTen VARCHAR(255),
    email VARCHAR(255),
    diaChi VARCHAR(255),
    soDienThoai VARCHAR(30),
    trangThai TINYINT(1),
    matKhau VARCHAR(255),
    facebookAccount VARCHAR(255),
    googleAccount VARCHAR(255),
    anhDaiDien VARCHAR(255),
    FOREIGN KEY (idVaiTro) REFERENCES VaiTro(id)
) ENGINE=InnoDB;

CREATE TABLE DanhMuc (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tenDanhMuc VARCHAR(255)
) ENGINE=InnoDB;

CREATE TABLE BaiDang (
    id INT AUTO_INCREMENT PRIMARY KEY,
    idNguoiDung INT,
    idDanhMuc INT,
    tieuDe VARCHAR(255),
    noiDung VARCHAR(255),
    ngayDang DATE,
    trangThai VARCHAR(255),
    lienHe VARCHAR(255),
    phuongThucThanhToan VARCHAR(255),
    FOREIGN KEY (idNguoiDung) REFERENCES NguoiDung(id),
    FOREIGN KEY (idDanhMuc) REFERENCES DanhMuc(id)
) ENGINE=InnoDB;

CREATE TABLE HinhAnhBaiDang (
    id INT AUTO_INCREMENT PRIMARY KEY,
    idBaiDang INT,
    loai VARCHAR(255),
    duongDan VARCHAR(255),
    thumbnailUrl VARCHAR(255),
    thuTu INT,
    FOREIGN KEY (idBaiDang) REFERENCES BaiDang(id)
) ENGINE=InnoDB;

CREATE TABLE ChiTietCanHo (
    id INT AUTO_INCREMENT PRIMARY KEY,
    idBaiDang INT,
    gia DOUBLE,
    dienTich FLOAT,
    phongNgu INT,
    diaChiCuThe VARCHAR(255),
    huongCanHo VARCHAR(255),
    ngayTao DATETIME,
    phuong VARCHAR(255),
    phongThuy VARCHAR(255),
    FOREIGN KEY (idBaiDang) REFERENCES BaiDang(id)
) ENGINE=InnoDB;

CREATE TABLE TienIch (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tenTienIch VARCHAR(255)
) ENGINE=InnoDB;

CREATE TABLE BaiDang_TienIch (
    idBaiDang INT,
    idTienIch INT,
    PRIMARY KEY (idBaiDang, idTienIch),
    FOREIGN KEY (idBaiDang) REFERENCES BaiDang(id),
    FOREIGN KEY (idTienIch) REFERENCES TienIch(id)
) ENGINE=InnoDB;

CREATE TABLE BaiDangYeuThich (
    idNguoiDung INT,
    idBaiDang INT,
    ngayTao DATETIME,
    PRIMARY KEY (idNguoiDung, idBaiDang),
    FOREIGN KEY (idNguoiDung) REFERENCES NguoiDung(id),
    FOREIGN KEY (idBaiDang) REFERENCES BaiDang(id)
) ENGINE=InnoDB;

CREATE TABLE TuongTacNguoiDung (
    id INT AUTO_INCREMENT PRIMARY KEY,
    idNguoiDung INT,
    idBaiDang INT,
    loaiHanhDong VARCHAR(255),
    ngayTao DATETIME,
    FOREIGN KEY (idNguoiDung) REFERENCES NguoiDung(id),
    FOREIGN KEY (idBaiDang) REFERENCES BaiDang(id)
) ENGINE=InnoDB;

CREATE TABLE NhuCauNguoiDung (
    id INT AUTO_INCREMENT PRIMARY KEY,
    idNguoiDung INT,
    minPrice DOUBLE,
    maxPrice DOUBLE,
    phuong VARCHAR(50),
    loaiCanHo VARCHAR(50),
    coBanCong TINYINT(1),
    dayDuNoiThat TINYINT(1),
    ganTrungTam TINYINT(1),
    ganBien TINYINT(1),
    ngayTao DATETIME,
    FOREIGN KEY (idNguoiDung) REFERENCES NguoiDung(id)
) ENGINE=InnoDB;

CREATE TABLE LichSuTimKiem (
    id INT AUTO_INCREMENT PRIMARY KEY,
    idNguoiDung INT,
    tuKhoa VARCHAR(255),
    minPrice DOUBLE,
    maxPrice DOUBLE,
    phuong VARCHAR(255),
    thoiGian DATETIME,
    FOREIGN KEY (idNguoiDung) REFERENCES NguoiDung(id)
) ENGINE=InnoDB;

CREATE TABLE PhuongThucThanhToan (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tenPhuongThucThanhToan VARCHAR(255),
    moTa VARCHAR(255)
) ENGINE=InnoDB;

CREATE TABLE MucDichThanhToan (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tenMucDich VARCHAR(255)
) ENGINE=InnoDB;

CREATE TABLE ChiTietThanhToan (
    id INT AUTO_INCREMENT PRIMARY KEY,
    idNguoiDung INT,
    idThanhToan INT,
    ngayBatDau DATE,
    ngayKetThuc DATE,
    soTien DOUBLE,
    trangThai VARCHAR(255),
    ngayTao DATETIME,
    FOREIGN KEY (idNguoiDung) REFERENCES NguoiDung(id)
) ENGINE=InnoDB;

CREATE TABLE LichSuGiaoDich (
    id INT AUTO_INCREMENT PRIMARY KEY,
    idNguoiDung INT,
    idThanhToan INT,
    soTien DOUBLE,
    trangThai VARCHAR(255),
    ngayTao DATETIME,
    noiDung VARCHAR(255),
    idMucDichThanhToan INT,
    FOREIGN KEY (idNguoiDung) REFERENCES NguoiDung(id),
    FOREIGN KEY (idThanhToan) REFERENCES ChiTietThanhToan(id),
    FOREIGN KEY (idMucDichThanhToan) REFERENCES MucDichThanhToan(id)
) ENGINE=InnoDB;
